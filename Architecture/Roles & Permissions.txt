# Grant Orchestrator permission to invoke the Provisioner service
gcloud run services add-iam-policy-binding iam-provisioner-service \
    --member="serviceAccount:${ORCHESTRATOR_SA}" \
    --role="roles/run.invoker" \
    --region=YOUR_REGION \
    --project=${PROJECT_ID}

# Grant Orchestrator access to Firestore (for session management)
gcloud projects add-iam-policy-binding ${PROJECT_ID} \
    --member="serviceAccount:${ORCHESTRATOR_SA}" \
    --role="roles/datastore.user"

# Grant Orchestrator access to Vertex AI (for NLU and RAG)
gcloud projects add-iam-policy-binding ${PROJECT_ID} \
    --member="serviceAccount:${ORCHESTRATOR_SA}" \
    --role="roles/aiplatform.user"

# Grant Orchestrator access to Discovery Engine (for RAG)
gcloud projects add-iam-policy-binding ${PROJECT_ID} \
    --member="serviceAccount:${ORCHESTRATOR_SA}" \
    --role="roles/discoveryengine.viewer"

# Provisioner needs minimal permissions (just logging for now)
gcloud projects add-iam-policy-binding ${PROJECT_ID} \
    --member="serviceAccount:${PROVISIONER_SA}" \
    --role="roles/logging.logWriter"

#Grant Orchestrator access to publish to iam-request-topic
gcloud pubsub topics add-iam-policy-binding iam-request-topic --member="${ORCHESTRATOR_SA}" --role="roles/pubsub.publisher"

#Create subscription to the iam-request-topic and grant access to Pub/Sub Invoker
gcloud pubsub subscriptions create iam-request-sub --topic iam-request-topic --push-endpoint=${ORCHESTRATOR SERVICE URL}/start_provisioning --push-auth-service-account=${Pub/Sub Invoker for IAM Orchestrator}

#Grant Pub/Sub Invoker to invoke orchestrator service
gcloud run services add-iam-policy-binding iam-orchestrator-service --region asia-southeast1 --member="${Pub/Sub Invoker for IAM Orchestrator}" --role="roles/run.invoker"

gcloud secrets create iam-jwt-secret --replication-policy="automatic"

echo -n "SUPER_SECRET_KEY_CHANGE_THIS" | \
    gcloud secrets versions add iam-jwt-secret --data-file=-
